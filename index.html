<script>

(function () {

    "use strict";

 

    var BASE = 67n;

    var CHARS = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz!@#$%";

    var charToVal = {};

    for (var i = 0; i < CHARS.length; i++) charToVal[CHARS[i]] = BigInt(i);

 

    var decInput = document.getElementById("decInput");

    var b67Input = document.getElementById("b67Input");

    var decHint = document.getElementById("decHint");

    var b67Hint = document.getElementById("b67Hint");

 

    var updating = false;

 

    function setHint(el, msg, isErr) {

        el.textContent = msg || "";

        el.className = "hint" + (isErr ? " error" : "");

    }

 

    // ---------- 十进制 -> 67进制 ----------

    function decTo67(decStr) {

        // 允许 +/- 号、空格、逗号

        var cleaned = decStr.replace(/[\s,]/g, "");

        if (cleaned === "") return { ok: true, value: "", neg: false };

        var m = cleaned.match(/^([+-]?)(\d+)$/);

        if (!m) return { ok: false, error: "无效的十进制输入" };

 

        var neg = m[1] === "-";

        var n = BigInt(m[2]);

        if (n === 0n) return { ok: true, value: "0", neg: false };

 

        var digits = "";

        while (n > 0n) {

            var r = n % BASE;

            digits = CHARS[Number(r)] + digits;

            n = n / BASE;

        }

        return { ok: true, value: digits, neg: neg && digits !== "0" };

    }

 

    // ---------- 67进制 -> 十进制 ----------

    function b67ToDec(b67Str) {

        var cleaned = b67Str.replace(/[\s,]/g, "");

        // 去掉可选前缀 67:（前缀前也可带符号）

        var m = cleaned.match(/^([+-]?)(?:67:)?(.*)$/i);

        var neg = m[1] === "-";

        var body = m[2];

        if (body === "") return { ok: true, value: "", neg: false };

 

        var n = 0n;

        for (var i = 0; i < body.length; i++) {

            var c = body[i];

            if (!(c in charToVal)) {

                return { ok: false, error: "非法字符: \"" + c + "\"" };

            }

            n = n * BASE + charToVal[c];

        }

        return { ok: true, value: n.toString(10), neg: neg && n !== 0n };

    }

 

    // ---------- 联动 ----------

    decInput.addEventListener("input", function () {

        if (updating) return;

        updating = true;

        var s = decInput.value;

        if (s.trim() === "") {

            b67Input.value = "";

            setHint(decHint, "");

            setHint(b67Hint, "");

            updating = false;

            return;

        }

        var r = decTo67(s);

        if (r.ok) {

            b67Input.value = r.value === "" ? "" : (r.neg ? "-" : "") + r.value;

            setHint(decHint, "");

            setHint(b67Hint, "");

        } else {

            setHint(decHint, r.error, true);

            setHint(b67Hint, "");

        }

        updating = false;

    });

 

    b67Input.addEventListener("input", function () {

        if (updating) return;

        updating = true;

        var s = b67Input.value;

        if (s.trim() === "") {

            decInput.value = "";

            setHint(decHint, "");

            setHint(b67Hint, "");

            updating = false;

            return;

        }

        var r = b67ToDec(s);

        if (r.ok) {

            decInput.value = r.value === "" ? "" : (r.neg ? "-" : "") + r.value;

            setHint(b67Hint, "");

            setHint(decHint, "");

        } else {

            setHint(b67Hint, r.error, true);

            setHint(decHint, "");

        }

        updating = false;

    });

 

    // ---------- 复制 / 清空 ----------

    function bindTool(btnId, input) {

        var btn = document.getElementById(btnId);

        if (btnId.indexOf("copy") === 0) {

            btn.addEventListener("click", function () {

                if (!input.value) return;

                var done = function () {

                    btn.textContent = "已复制";

                    setTimeout(function () { btn.textContent = "复制"; }, 1200);

                };

                if (navigator.clipboard && navigator.clipboard.writeText) {

                    navigator.clipboard.writeText(input.value).then(done, done);

                } else {

                    input.select();

                    document.execCommand("copy");

                    done();

                }

            });

        } else {

            btn.addEventListener("click", function () {

                input.value = "";

                input.dispatchEvent(new Event("input"));

                input.focus();

            });

        }

    }

    bindTool("copy10", decInput);

    bindTool("clear10", decInput);

    bindTool("copy67", b67Input);

    bindTool("clear67", b67Input);

 

    // ---------- 字符对照表 ----------

    var tbody = document.getElementById("tableBody");

    var cols = 4, rows = Math.ceil(67 / cols);

    var html = "";

    for (var r = 0; r < rows; r++) {

        html += "<tr>";

        for (var c = 0; c < cols; c++) {

            var v = r + c * rows;

            if (v < 67) {

                html += '<td class="num">' + v + '</td><td class="chr">' +

                        (CHARS[v] === "<" ? "&lt;" : CHARS[v] === ">" ? "&gt;" : CHARS[v] === "&" ? "&amp;" : CHARS[v]) +

                        "</td>";

            } else {

                html += '<td class="num"></td><td class="chr"></td>';

            }

        }

        html += "</tr>";

    }

    tbody.innerHTML = html;

})();

</script>
